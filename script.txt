// ---------- कॉलम मैपिंग हेल्पर ----------
let _headers = null;
let _headerRow = null;

function getHeaders() {
  if (_headers) return _headers;
  const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME);
  const data = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  _headers = data.map(h => String(h).trim());
  _headerRow = 1;
  return _headers;
}

function getCol(headerName) {
  const headers = getHeaders();
  const idx = headers.indexOf(headerName);
  if (idx === -1) throw new Error('Column "' + headerName + '" not found in sheet.');
  return idx + 1; // 1‑based
}

function setValueInRow(rowArray, headerName, value) {
  const col = getCol(headerName);
  rowArray[col - 1] = value; // rowArray 0‑based
}

function appendRowWithHeaders(rowArray) {
  const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME);
  const headers = getHeaders();
  while (rowArray.length < headers.length) rowArray.push('');
  sheet.appendRow(rowArray);
}

// ==================== doGet – Receipt Lookup ====================
function doGet(e) {
  try {
    const action = e.parameter.action;
    if (action === 'lookupReceipt') {
      const paymentId = (e.parameter.paymentId || '').trim();
      const pan = (e.parameter.pan || '').trim().toUpperCase();
      const mobile = (e.parameter.mobile || '').trim();

      const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME);
      const data = sheet.getDataRange().getValues();
      let found = null;

      for (let i = 1; i < data.length; i++) {
        const row = data[i];
        const rowPaymentId = String(row[getCol('Payment ID') - 1] || '').trim();
        const rowPan = String(row[getCol('PAN Number') - 1] || '').trim().toUpperCase();
        const rowMobile = String(row[getCol('Phone') - 1] || '').trim();
        const rowReceiptUrl = String(row[getCol('Receipt PDF Link') - 1] || '').trim();

        if ((paymentId && rowPaymentId === paymentId) ||
            (pan && rowPan === pan) ||
            (mobile && rowMobile === mobile)) {
          found = {
            paymentId: rowPaymentId,
            donorName: String(row[getCol('First Name / Name') - 1] || '').trim(),
            receiptUrl: rowReceiptUrl
          };
          break;
        }
      }

      if (found) {
        return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: found }))
          .setMimeType(ContentService.MimeType.JSON);
      } else {
        return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Not found' }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Invalid action' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function createResponse(status, message) {

}

function getOrCreateFolder(name) {
  const folders = DriveApp.getFoldersByName(name);
  return folders.hasNext() ? folders.next() : DriveApp.createFolder(name);
}

function saveFile(base64Data, fileName, folder) {
  if (!base64Data) return '';
  const matches = base64Data.match(/^data:([A-Za-z-+\/]+);base64,(.+)$/);
  if (!matches) return '';
  const blob = Utilities.newBlob(Utilities.base64Decode(matches[2]), matches[1], fileName);
  const file = folder.createFile(blob);
  return file.getUrl();
}

function generateVolunteerId() {
  const now = new Date();
  const y = String(now.getFullYear()).slice(-2);
  const m = String(now.getMonth()+1).padStart(2,'0');
  const d = String(now.getDate()).padStart(2,'0');
  return 'VOL' + y + m + d + String(Math.floor(100 + Math.random() * 900));
}

function escapeHtml(str) {
  if (!str) return '';
  return String(str).replace(/[&<>]/g, function(m) {
    if (m === '&') return '&amp;';
    if (m === '<') return '&lt;';
    if (m === '>') return '&gt;';
    return m;
  });
}
